function segDist(x,y,a,b){const dx=b[0]-a[0],dy=b[1]-a[1],q=dx*dx+dy*dy||1;let t=((x-a[0])*dx+(y-a[1])*dy)/q;t=Math.max(0,Math.min(1,t));return Math.hypot(a[0]+t*dx-x,a[1]+t*dy-y);}
function nearest(x,y){let best=null;const R=14/Math.max(1,view.s*.8);
  SWITCHES.forEach(sw=>{const d=Math.hypot(sw.pos[0]-x,sw.pos[1]-y);if(d<R*1.2&&(!best||d<best.d))best={d,ref:sw.id,label:'Switch '+sw.id+' ('+sw.name+')'};});
  if(best)return best;
  EN.forEach(e=>{const d=Math.hypot(e.pos[0]-x,e.pos[1]-y);if(d<R*1.2&&(!best||d<best.d))best={d,ref:e.id,label:e.name};});if(best)return best;
  NET.gates.forEach(([gx,gy],i)=>{const d=Math.hypot(gx-x,gy-y);if(d<R*.8&&(!best||d<best.d))best={d,ref:'G'+(i+1),label:'Puerta'};});if(best)return best;
  ALL.forEach(k=>{const p=L[k].pts;for(let i=0;i<p.length-1;i++){const d=segDist(x,y,p[i],p[i+1]);if(d<R&&(!best||d<best.d))best={d,ref:k,label:'Línea: '+L[k].name};}});
  return best;}

  el('text',{y:-13,'text-anchor':'middle',class:'pin-n',fill:kind==='draft'?'#B01F68':'#fff'},g).textContent=n;return g;}

/* respaldo local con la misma forma que la base de datos */
function localStore(){const KEY='cerco-correcciones-v1';let docs={},subs=[],seq=0;
  try{docs=JSON.parse(localStorage.getItem(KEY)||'{}')||{};}catch(e){docs={};}
  const save=()=>{try{localStorage.setItem(KEY,JSON.stringify(docs));}catch(e){}};
  const snap=()=>({metadata:{fromCache:false},docs:Object.keys(docs).map(id=>({id,data:()=>docs[id]}))});
  const fire=()=>subs.forEach(f=>f(snap()));
  return {collection:()=>({onSnapshot(n){subs.push(n);setTimeout(()=>n(snap()),0);return()=>{};},
      add:async b=>{const id='l'+Date.now().toString(36)+(seq++);docs[id]=b;save();fire();return {id};}}),
    doc:p=>{const id=p.split('/')[1];return {set:async b=>{docs[id]=b;save();fire();},delete:async()=>{delete docs[id];save();fire();}};}};}
function startNotes(db){if(!db){db=localStore();LOCAL=true;}DB=db;
  [markBtn,markBtn2].forEach(b=>b.disabled=false);
  const timer=setTimeout(()=>{fixStatus.textContent='Todavía no se pueden cargar las correcciones guardadas; puede que no estén al día.';},10000);
  db.collection('correcciones').onSnapshot(snap=>{if(!snap.metadata.fromCache)clearTimeout(timer);
      notes=snap.docs.map(d=>Object.assign({id:d.id},d.data())).sort((a,b)=>(a.n||0)-(b.n||0));
      if(editing&&!notes.find(n=>n.id===editing.id))closeForm();
      renderNotes();statusIdle();},
    err=>{clearTimeout(timer);fixStatus.textContent='No se pudieron cargar las correcciones. Vuelve a abrir el mapa más tarde.';});}
